/* =========================================================
   🎵 CUTE BACKGROUND MUSIC
   Browser click नंतरच music सुरू करतो.
========================================================= */

let audioStarted = false;

let audioContext;

let masterGain;


function startMusic() {

    if(audioStarted) return;

    audioStarted = true;


    try {

        audioContext =
            new (
                window.AudioContext ||
                window.webkitAudioContext
            )();


        masterGain =
            audioContext.createGain();

        masterGain.gain.value =
            0.045;

        masterGain.connect(
            audioContext.destination
        );


        /*
           Soft little melody
        */

        const notes = [
            261.63,
            329.63,
            392.00,
            329.63,
            293.66,
            349.23,
            440.00,
            349.23
        ];


        let index = 0;


        function playNote() {

            const oscillator =
                audioContext.createOscillator();

            const gain =
                audioContext.createGain();


            oscillator.type =
                "sine";

            oscillator.frequency.value =
                notes[index];


            oscillator.connect(gain);

            gain.connect(masterGain);


            const now =
                audioContext.currentTime;


            gain.gain.setValueAtTime(
                0,
                now
            );

            gain.gain.linearRampToValueAtTime(
                0.35,
                now + .08
            );

            gain.gain.exponentialRampToValueAtTime(
                0.001,
                now + 1.1
            );


            oscillator.start(now);

            oscillator.stop(
                now + 1.2
            );


            index =
                (index + 1) % notes.length;

        }


        playNote();


        setInterval(
            playNote,
            850
        );


    } catch(error) {

        console.log(
            "Music unavailable",
            error
        );

    }

}
